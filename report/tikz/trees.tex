% tikz/trees.tex -- figure: the tree topologies (input from sections/02_setup.tex).
% Node layout follows ttn/tree.py: binary_tree, mps_tree (root = last core), ht_tree; star3 and midroot6
% as in experiments/locality/make_configs.py. Sites and cores are numbered from 0, as in the code.
\tikzset{
  ttn/core/.style={circle, draw=MidnightBlue!75!black, fill=MidnightBlue!12, line width=0.6pt,
                   minimum size=3.2mm, inner sep=0pt},
  ttn/root/.style={ttn/core, fill=MidnightBlue!60},
  ttn/bond/.style={line width=1.1pt, black!80},
  ttn/phys/.style={line width=0.6pt, black!55},
  ttn/site/.style={font=\scriptsize, inner sep=1pt, text=black!75},
}
\begin{figure}[t]
  \centering
  \begin{subfigure}[b]{0.21\textwidth}
    \centering
    \begin{tikzpicture}[x=4.2mm, y=7mm]
      \foreach \j in {0,...,3} \node[ttn/core] (b\j) at ({2*\j+1}, 0) {};
      \node[ttn/core] (m0) at (2, 1) {};
      \node[ttn/core] (m1) at (6, 1) {};
      \node[ttn/root] (r) at (4, 2) {};
      \draw[ttn/bond] (b0) -- (m0) -- (b1)  (b2) -- (m1) -- (b3)  (m0) -- (r) -- (m1);
      \foreach \j in {0,...,3} {
        \pgfmathtruncatemacro{\sa}{2*\j}
        \pgfmathtruncatemacro{\sb}{2*\j+1}
        \draw[ttn/phys] (b\j) -- ++(-0.45,-0.8) node[ttn/site, below] {\sa};
        \draw[ttn/phys] (b\j) -- ++(0.45,-0.8) node[ttn/site, below] {\sb};
      }
    \end{tikzpicture}
    \caption{binary8, $\Kc=7$}
    \label{fig:trees-binary}
  \end{subfigure}%
  \hfill
  \begin{subfigure}[b]{0.21\textwidth}
    \centering
    \begin{tikzpicture}[x=4.2mm, y=7mm]
      \foreach \k in {0,...,6} \node[ttn/core] (c\k) at (\k, 0) {};
      \node[ttn/root] (c7) at (7, 0) {};
      \foreach \k [evaluate=\k as \kn using int(\k+1)] in {0,...,6}
        \draw[ttn/bond] (c\k) -- (c\kn);
      \foreach \k in {0,...,7}
        \draw[ttn/phys] (c\k) -- ++(0,-0.8) node[ttn/site, below] {\k};
      \node[font=\scriptsize, above=1.5pt] at ($(c3)!0.5!(c4)$) {$\bd{3}$};
    \end{tikzpicture}
    \caption{MPS8, $\Kc=8$}
    \label{fig:trees-mps}

  \hfill
  \begin{subfigure}[b]{0.16\textwidth}

    \begin{tikzpicture}[x=5mm, y=7mm]

      \node[ttn/core] (h0) at (0.5, 1) {};
      \node[ttn/core] (h1) at (2.5, 1) {};
      \node[ttn/root] (r) at (1.5, 2) {};

        \draw[ttn/phys] (l\i) -- ++(0,-0.8) node[ttn/site, below] {\i};
    \end{tikzpicture}
    \caption{HT4, $\Kc=7$}
    \label{fig:trees-ht}
  \end{subfigure}%
  \hfill
  \begin{subfigure}[b]{0.17\textwidth}
    \centering
    \begin{tikzpicture}[x=4.2mm, y=7mm]

        \pgfmathtruncatemacro{\sa}{2*\j}

        \draw[ttn/phys] (s\j) -- ++(0.45,-0.8) node[ttn/site, below] {\sb};

  \caption{The trees of \cref{tab:trees}. Circles are cores $\G{v}$; the shaded core is the root,

    $\edge{v}$ between core $v$ and its parent has dimension $\bd{v}$ (one label shown in (b), where
    core $i$ carries site $i$). Thin dangling lines are physical legs, labelled by their site $i$
    (dimension $\pd[i]$). (d) and (e) are the two further trees of the locality experiments
    (\cref{sec:s2s-locality}).}
  \label{fig:trees}
